def write_config(text):
    """Zapise TOML do docasneho souboru a vrati jeho cestu."""
    tmp = tempfile.NamedTemporaryFile("w", suffix=".toml", delete=False, encoding="utf-8")
    tmp.write(text)
    tmp.close()
    return tmp.name


class ConfigPathTest(unittest.TestCase):
    def test_uses_xdg_config_home(self):
        with mock.patch.dict(os.environ, {"XDG_CONFIG_HOME": "/tmp/xdgtest"}):
            self.assertEqual(config_path(), Path("/tmp/xdgtest/sysmon-app/config.toml"))

    def test_relative_xdg_is_ignored(self):
        """XDG basedir spec: relativni cesta se zahazuje."""
        with mock.patch.dict(os.environ, {"XDG_CONFIG_HOME": "relativni"}):
            self.assertTrue(config_path().is_absolute())


class LoadSettingsTest(unittest.TestCase):
    def test_missing_file_gives_defaults(self):
        s = load_settings("/tmp/sysmon-app-neexistuje.toml")
        self.assertEqual(s.port, DEFAULT_PORT)
        self.assertEqual(s.watch_ports, DEFAULT_WATCH_PORTS)

    def test_reads_host_and_port(self):
        s = load_settings(write_config('host = "127.0.0.1"\nport = 9000\n'))
        self.assertEqual(s.host, "127.0.0.1")
        self.assertEqual(s.port, 9000)

    def test_watch_ports_keys_become_ints(self):
        s = load_settings(write_config('[watch_ports]\n22 = "SSH"\n8080 = "Backend"\n'))
        self.assertEqual(s.watch_ports, {22: "SSH", 8080: "Backend"})

    def test_ignored_ports_become_ints(self):
        s = load_settings(write_config("ignored_ports = [53, 631]\n"))
        self.assertEqual(s.ignored_ports, {53, 631})

    def test_unknown_key_is_an_error(self):
        """Preklep v klíči nesmi tise nic nedelat."""
        path = write_config("prot = 9000\n")
        with self.assertRaises(ValueError) as ctx:
            load_settings(path)
        self.assertIn("prot", str(ctx.exception))
        self.assertIn(path, str(ctx.exception))

    def test_broken_toml_names_the_path(self):
        path = write_config("port = \n")
        with self.assertRaises(ValueError) as ctx:
            load_settings(path)
        self.assertIn(path, str(ctx.exception))

    def test_watch_ports_must_be_a_table(self):
        path = write_config("watch_ports = [22, 3306]\n")
        with self.assertRaises(ValueError) as ctx:
            load_settings(path)
        self.assertIn("watch_ports", str(ctx.exception))

    def test_ignored_ports_must_be_a_list(self):
        """Tabulka i retezec jsou chyba - "53" by se tise rozpadlo na {3, 5}."""
        for text in ("[ignored_ports]\n53 = 1\n", 'ignored_ports = "53"\n',

                path = write_config(text)

                self.assertIn("ignored_ports", str(ctx.exception))

        for text in ("port = true\n", "ignored_ports = [true]\n"):
            with self.subTest(text=text):
                with self.assertRaises(ValueError):
                    load_settings(write_config(text))

"""Konfigurace sysmon-app: vychozi hodnoty a volitelny config.toml od uzivatele.

Stroj-specificke veci (ktere porty sledovat, na cem poslouchat) nepatri do
kodu - na cizim stroji by byly nesmysl. Vsechno ostatni jsou konstanty.
"""

import os
import tomllib
from dataclasses import dataclass, field
from pathlib import Path

DEFAULT_HOST = "0.0.0.0"
DEFAULT_PORT = 8430          # sysmon ma 8420; oba produkty zijou vedle sebe

TICK_SEC = 1.5               # perioda sberneho cyklu = perioda SSE
SSE_KEEPALIVE_SEC = 20.0     # po teto dobe ticha posleme komentar, at spojeni nezivori

DOCKER_POLL_SEC = 5
BATTERY_HISTORY_SEC = 120
BATTERY_HISTORY_MAX = 720    # 24 h pri 2min intervalu
TOP_PROCESSES_N = 5

DEFAULT_WATCH_PORTS = {
    22: "SSH",
    3306: "MySQL",
    5173: "Vite",
    8080: "Backend",
    8081: "Metro",
}
DEFAULT_IGNORED_PORTS = {53, 111, 631, 5353, 1716, 54321}

CONFIG_DIR_NAME = "sysmon-app"
CONFIG_FILE_NAME = "config.toml"

_KNOWN_KEYS = {"host", "port", "watch_ports", "ignored_ports"}


def config_path():
    """Cesta ke konfiguraci. Respektuje XDG_CONFIG_HOME, relativni ignoruje.

    Relativni XDG_CONFIG_HOME by znamenal, ze cesta k tokenu zavisi na
    aktualnim adresari procesu - to XDG basedir spec zakazuje.
    """
    default = os.path.join(os.path.expanduser("~"), ".config")
    base = os.environ.get("XDG_CONFIG_HOME") or default
    if not os.path.isabs(base):
        base = default
    return Path(base) / CONFIG_DIR_NAME / CONFIG_FILE_NAME


@dataclass
class Settings:
    host: str = DEFAULT_HOST
    port: int = DEFAULT_PORT
    watch_ports: dict = field(default_factory=lambda: dict(DEFAULT_WATCH_PORTS))
    ignored_ports: set = field(default_factory=lambda: set(DEFAULT_IGNORED_PORTS))


def _port_number(value, key):
    """Prevede klic/hodnotu na cislo portu, nebo rekne, co je spatne.

    bool se odmita zvlast: je to potomek int, takze `port = true` by tise
    znamenalo port 1. Stejna past, jakou v sysmonu resi validate_pid.
    """
    if isinstance(value, bool):
        raise ValueError(f"{key}: '{value}' není číslo portu")
    try:
        port = int(value)
    except (TypeError, ValueError):
        raise ValueError(f"{key}: '{value}' není číslo portu") from None
    if not 1 <= port <= 65535:
        raise ValueError(f"{key}: port {port} je mimo rozsah 1-65535")
    return port


def load_settings(path=None):
    """Nacte Settings. Chybejici soubor znamena vychozi hodnoty.

    Rozbita konfigurace se nepolyka: tise ignorovany preklep by uzivateli
    vzal jeho nastaveni, aniž by se dozvedel proc.
    """
    path = Path(path) if path is not None else config_path()
    if not path.is_file():
        return Settings()

    try:
        with open(path, "rb") as f:
            raw = tomllib.load(f)
    except tomllib.TOMLDecodeError as exc:
        raise ValueError(f"Konfigurace {path} není platné TOML: {exc}") from None
    except OSError as exc:
        raise ValueError(f"Konfiguraci {path} nelze přečíst: {exc}") from None

    unknown = sorted(set(raw) - _KNOWN_KEYS)
    if unknown:
        raise ValueError(
            f"Konfigurace {path} obsahuje neznámé klíče: {', '.join(unknown)}. "
            f"Povolené jsou: {', '.join(sorted(_KNOWN_KEYS))}."
        )

    settings = Settings()
    if "host" in raw:
        settings.host = str(raw["host"])
    if "port" in raw:
        settings.port = _port_number(raw["port"], "port")
    if "watch_ports" in raw:
        table = raw["watch_ports"]
        if not isinstance(table, dict):
            raise ValueError(
                f"Konfigurace {path}: watch_ports musí být tabulka "
                '[watch_ports] s porty jako klíči, např. 22 = "SSH".'
            )
        settings.watch_ports = {
            _port_number(port, f"watch_ports.{port}"): str(label)
            for port, label in table.items()
        }
    if "ignored_ports" in raw:
        ports = raw["ignored_ports"]
        if not isinstance(ports, list):
            raise ValueError(
                f"Konfigurace {path}: ignored_ports musí být seznam čísel portů, "
                "např. ignored_ports = [53, 631]."
            )
        settings.ignored_ports = {
            _port_number(port, "ignored_ports") for port in ports
        }
    return settings
```

LISTEN_22 = ("   0: 0100007F:0016 00000000:0000 0A 00000000:00000000 00:00000000 "

class ListeningPortsTest(unittest.TestCase):
    def _fake_tcp(self, content):
        fd, path = tempfile.mkstemp(prefix="faketcp")
        with os.fdopen(fd, "w") as f:
            f.write("  sl  local_address rem_address   st\n" + content)
        self.addCleanup(os.unlink, path)
        orig = ports.TCP_FILES
        ports.TCP_FILES = (path,)
        self.addCleanup(setattr, ports, "TCP_FILES", orig)

    def test_parses_hex_port(self):
        self._fake_tcp(LISTEN_22)
        self.assertEqual(ports.listening_ports(set()), {22})

    def test_ignores_non_listen_states(self):
        self._fake_tcp(ESTAB_8080)
        self.assertEqual(ports.listening_ports(set()), set())

    def test_ignores_ignored_ports(self):
        self._fake_tcp(LISTEN_53)
        self.assertEqual(ports.listening_ports({53}), set())

    def test_missing_file_is_not_fatal(self):
        orig = ports.TCP_FILES
        ports.TCP_FILES = ("/neexistuje/tcp",)

"""Naslouchajici porty - cte se primo z /proc/net/tcp{,6}, bez shellu.

Proc ne `ss`/`netstat`: zbytecny podproces, a tyhle soubory jsou autoritativni.

LISTEN_STATE = "0A"


def listening_ports(ignored_ports):
    """Mnozina cisel portu, ktere nekdo posloucha (bez ignored_ports)."""
    ports = set()
    for path in TCP_FILES:
        try:
            with open(path) as f:
                next(f)
                for line in f:
                    parts = line.split()
                    if len(parts) < 4 or parts[3] != LISTEN_STATE:
                        continue
                    try:
                        port_hex = parts[1].rsplit(":", 1)[1]
                        port = int(port_hex, 16)
                    except (IndexError, ValueError):
                        continue
                    if port in ignored_ports:
                        continue
                    ports.add(port)
        except (OSError, StopIteration):
            continue
    return ports


def get_services(watch_ports, ignored_ports):
    """Sledovane porty - nejdriv bezi, pak offline, vzestupne.

    Port, ktery neni ve watch_ports, je neviditelny; watch_ports je jediny
    zdroj pravdy o tom, co karta sluzeb ukazuje.
    """
    listening = listening_ports(ignored_ports)
    items = [
        {"port": port, "label": label, "on": port in listening}
        for port, label in watch_ports.items()
    ]
    items.sort(key=lambda x: (not x["on"], x["port"]))
    return items
```

```python
"""SnapshotFeed - jeden sberny cyklus pro vsechny odberatele.

Proc takhle: driv cetl kolektory kazdy HTTP pozadavek z vlastniho vlakna,
takze se rate-based kolektory (sit, disk) navzajem rozbilyly - v sysmonu
zapsana a prijata vada. Tady tiká jediny cyklus a odberatele se jen vezou.

Odberatel se prihlasi tim, ze si zapamatuje `snapshot_seq()` a pak ceka
`wait_for(seq, timeout)`. Zadny fronta k odberateli, zadne kopirovani
snapshotu pro kazdeho - vsichni dostanou ten sam dict (jen se v nem nemeni).
"""

import threading

from sysmon_app.config import TICK_SEC


class SnapshotFeed:
    def __init__(self, monitor, interval=TICK_SEC):
        self._monitor = monitor
        self._interval = interval
        self._cond = threading.Condition()
        self._seq = 0
        self._latest = None
        self._stop = threading.Event()
        self._thread = None

    def publish_once(self):
        """Sebrat jeden snapshot a rozeslat ho odberatelum.

        Vzorkovani je zamerne mimo zamek - drzet zamek pres cteni /proc
        a `docker stats` by zastavilo vsechny odberatele.
        """
        snapshot = self._monitor.snapshot()
        with self._cond:
            self._latest = snapshot
            self._seq += 1
            self._cond.notify_all()

    def latest(self):
        """Posledni snapshot, nebo None kdyz jeste zadny nebyl."""
        with self._cond:
            return self._latest

    def snapshot_seq(self):
        with self._cond:
            return self._seq

    def wait_for(self, last_seq, timeout):
        """Pocka na snapshot novejsi nez last_seq. Vraci (seq, snapshot).

        Po timeoutu vraci (last_seq, None) - volatel to bere jako "nic
        noveho", ne jako chybu.
        """
        with self._cond:
            self._cond.wait_for(
                lambda: self._seq != last_seq or self._stop.is_set(), timeout,
            )
            if self._seq == last_seq:
                return last_seq, None
            return self._seq, self._latest

    def start(self):
        if self._thread is not None:
            return
        self._thread = threading.Thread(
            target=self._loop, name="sysmon-app-feed", daemon=True,
        )
        self._thread.start()

    def stop(self):
        self._stop.set()
        with self._cond:
            self._cond.notify_all()

    def join(self, timeout=None):
        if self._thread is not None:
            self._thread.join(timeout)

    def is_running(self):
        return self._thread is not None and self._thread.is_alive()

    def _loop(self):
        # Prvni sber hned: jinak by prvni pozadavek cekal cely TICK_SEC.
        while not self._stop.is_set():
            try:
                self.publish_once()
            except Exception:
                # Sber se nesmi rozbit smycku - dalsi tik to zkusi znovu.
                pass
            self._stop.wait(self._interval)
```

```python
"""HTTP server - routy, SPA a SSE proud.

Routy:
    GET /              -> web/index.html
    GET /static/<soubor> -> web/static/style.css, app.js, icon.svg
    GET /stats         -> JSON snapshot
    GET /events        -> SSE proud snapshotu

Zadna autentizace a zadne POST routy: produkt nic nemeni, takze neni co
chranit. Pozor ale, ze server defaultne posloucha na vsech rozhranich -
kdokoli v siti vidi seznam procesu a portu. Je to vedome, kvuli telefonu.
"""

import json
import os
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

from sysmon_app.config import DEFAULT_HOST, DEFAULT_PORT, SSE_KEEPALIVE_SEC

WEB_DIR = Path(__file__).resolve().parent / "web"
STATIC_DIR = WEB_DIR / "static"
INDEX_NAME = "index.html"

STATIC_PREFIX = "/static/"

# Povolene pripony pro /static/ -> content-type
STATIC_TYPES = {
    ".css": "text/css; charset=utf-8",
    ".js": "application/javascript; charset=utf-8",
    ".svg": "image/svg+xml",
}


def load_index():
    """Nacte web/index.html (UTF-8)."""
    with open(WEB_DIR / INDEX_NAME, encoding="utf-8") as f:
        return f.read()


def load_static(name):
    """Nacte soubor z web/static/. Vraci (bytes, content-type) nebo None."""
    ctype = STATIC_TYPES.get(os.path.splitext(name)[1])
    if ctype is None:
        return None
    real = os.path.realpath(os.path.join(STATIC_DIR, name))
    root = os.path.realpath(STATIC_DIR) + os.sep
    if not real.startswith(root):  # ochrana proti vyskoku z adresare pres ".."
        return None
    try:
        with open(real, "rb") as f:
            return f.read(), ctype
    except OSError:
        return None


class Handler(BaseHTTPRequestHandler):
    # HTTP/1.1 kvuli SSE: u HTTP/1.0 klient nepozna konec tela jinak nez
    # zavrenim spojeni, takze by se EventSource porad pripojoval znovu.
    protocol_version = "HTTP/1.1"

    # Nastavi serve() pred spustenim serveru.
    monitor = None
    feed = None

    def _path(self):
        """Cesta bez query stringu - routy se porovnavaji presne."""
        return self.path.partition("?")[0]

    def _send(self, body, ctype, status=200):
        data = body.encode() if isinstance(body, str) else body
        self.send_response(status)
        self.send_header("Content-Type", ctype)
        self.send_header("Content-Length", str(len(data)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(data)

    def _send_json(self, obj, status=200):
        self._send(json.dumps(obj), "application/json; charset=utf-8", status=status)

    def _send_empty(self, status):
        """Odpoved bez tela. Content-Length 0 je nutny: s HTTP/1.1 by klient
        jinak cekal na telo, ktere neprijde."""
        self.send_response(status)
        self.send_header("Content-Length", "0")
        self.end_headers()

    def do_GET(self):
        path = self._path()
        if path == "/stats":
            self._send_json(self._snapshot_now())
        elif path in ("/", "/index.html"):
            self._send(load_index(), "text/html; charset=utf-8")
        elif path.startswith(STATIC_PREFIX):
            asset = load_static(path[len(STATIC_PREFIX):])
            if asset is None:
                self._send_empty(404)
            else:
                self._send(asset[0], asset[1])
        else:
            self._send_empty(404)

    def _snapshot_now(self):
        """Snapshot z feedu; kdyz jeste zadny nebyl, sebere se naprimo.

        Diky tomu /stats nikdy necte kolektory samo, jakmile feed bezi -
        jinak by se rate-based kolektory rozbilyly soubeznym ctenim.
        """
        snapshot = self.feed.latest()
        if snapshot is None:
            snapshot = self.monitor.snapshot()
        return snapshot

    def log_message(self, *args):
        pass


def serve(monitor, feed, host=DEFAULT_HOST, port=DEFAULT_PORT):
    """Vytvori server s predanym Monitorem a feedem (zadne globalni promenne)."""
    Handler.monitor = monitor
    Handler.feed = feed
    srv = ThreadingHTTPServer((host, port), Handler)
    # SSE spoje drzi vlakno kazdy. Bez daemon_threads by ukonceni serveru
    # cekalo na ne, dokud je klient nezatvori.
    srv.daemon_threads = True
    return srv
```

import json
import socket
import threading

def read_headers(f):
    """Precte hlavicky do prazdneho radku."""
    lines = []
    while True:
        line = f.readline()
        if line in (b"\r\n", b"\n", b""):
            return lines
        lines.append(line.decode("latin-1"))


def read_chunk(f):
    """Precte jedno chunked telo. Vraci bytes."""
    size = int(f.readline().strip().split(b";")[0], 16)
    body = f.read(size)
    f.read(2)  # CRLF za telem
    return body


class SseTest(unittest.TestCase):
    def setUp(self):
        monitor = FakeMonitor()
        feed = SnapshotFeed(monitor, interval=0.05)
        feed.start()
        self.addCleanup(feed.stop)
        self.server = server.serve(monitor, feed, "127.0.0.1", 0)
        self.addCleanup(self.server.server_close)
        threading.Thread(target=self.server.serve_forever, daemon=True).start()
        self.addCleanup(self.server.shutdown)
        self.port = self.server.server_address[1]

    def open_stream(self):
        sock = socket.create_connection(("127.0.0.1", self.port), timeout=5)
        self.addCleanup(sock.close)
        sock.sendall(b"GET /events HTTP/1.1\r\nHost: 127.0.0.1\r\n\r\n")
        f = sock.makefile("rb")
        self.addCleanup(f.close)
        return f

    def test_headers_announce_event_stream(self):

        self.assertIn("text/event-stream", headers)
        self.assertIn("chunked", headers.lower())

        read_headers(f)
        body = read_chunk(f)

        f = self.open_stream()
        read_headers(f)
        read_chunk(f)

```python
    def _stream_events(self):
        """Posila snapshoty, dokud klient neodejde.

        Framovani je chunked rucne: u SSE se neda poslat Content-Length,
        protoze telo nema konec. Kazdy snapshot je jeden chunk.
        """
        self.close_connection = True
        self.send_response(200)
        self.send_header("Content-Type", "text/event-stream; charset=utf-8")
        self.send_header("Cache-Control", "no-store")
        self.send_header("Transfer-Encoding", "chunked")
        self.end_headers()

        last_seq = self.feed.snapshot_seq()
        try:
            while True:
                seq, snapshot = self.feed.wait_for(last_seq, timeout=SSE_KEEPALIVE_SEC)
                if snapshot is None:
                    # Nic noveho - komentarovy ramec drzi spojeni nazivu.
                    self._write_chunk(b": keep-alive\n\n")
                    continue
                last_seq = seq
                payload = json.dumps(snapshot).encode("utf-8")
                self._write_chunk(b"data: " + payload + b"\n\n")
        except (BrokenPipeError, ConnectionResetError, OSError):
            # Klient zavrel okno nebo usel - to neni chyba serveru.
            return

    def _write_chunk(self, data):
        self.wfile.write(b"%x\r\n" % len(data) + data + b"\r\n")
        self.wfile.flush()
```

Import `SSE_KEEPALIVE_SEC` už v souboru je z Tasku 9.

def parse_args(argv=None, settings=None):
    settings = settings or load_settings()
    ap = argparse.ArgumentParser(
        prog="sysmon_app",
        description="Dashboard systémových metrik pro okno i telefon.",
    )
    ap.add_argument("--host", default=settings.host,

def main(argv=None):

    except ValueError as exc:
        # Rozbita konfigurace se hlasi vetou, ne tracebackem - appku pouzivaji

k ověření démona — kontrakt snapshotu se nemění, takže stará UI musí fungovat
proti novému démonu. Fáze 2 ji nahradí Reactem.

```bash

sleep 3

# stejná čísla z obou?
curl -s localhost:8420/stats | python3 -c "import json,sys; d=json.load(sys.stdin); print('sysmon   ', d['host'], round(d['cpu'],1), d['mem']['percent'], len(d['processes']), len(d['services']))"
curl -s localhost:8430/stats | python3 -c "import json,sys; d=json.load(sys.stdin); print('sysmon-app', d['host'], round(d['cpu'],1), d['mem']['percent'], len(d['processes']), len(d['services']))"
```
Expected: stejný hostname, srovnatelné CPU a paměť, stejný počet procesů. Počet služeb se může **lišit**, pokud jsi upravil `watch_ports` — výchozí seznam je stejný jako v `sysmonu`, takže bez konfigurace musí sedět.

Pak otevři v prohlížeči `http://localhost:8430` a porovnej karty s `http://localhost:8420` okem: CPU jádra, paměť, swap, disk, síť, teploty, baterie, služby, kontejnery. Musí odpovídat.
