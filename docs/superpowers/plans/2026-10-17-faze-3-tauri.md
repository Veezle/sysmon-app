> **Pro pracovníky-agenty:** povinný sub-skill:
> `superpowers-extended-cc:subagent-driven-development` (doporučeno), nebo

**Goal:** Postavit Tauri 2 obal kolem hotového démona: okno na dashboard, tray,
dozor nad sidecarem, PDEATHSIG proti osiřelému démonovi, autostart a jedna

**Architecture:** `src-tauri/` je Rust aplikace bez vlastního frontendu;
`frontendDist` je externí URL `http://127.0.0.1:8430/`, takže se do binárky
nevkládá žádný React. Tauri spouští `python3 -m sysmon_app` jako potomka,
čeká na port, dohlíží na něj a při pádu aplikace ho nechá zemřít přes
`PR_SET_PDEATHSIG`.

**Tech Stack:** Tauri 2 (Rust 1.95), WebKitGTK 4.1, gtk-rs 0.18 (nativní
dialogy), `tauri-plugin-single-instance`, `libc` (PDEATHSIG). Python a Node
zůstávají beze změny.

```rust
use tauri::{AppHandle, Manager, WebviewUrl, WebviewWindowBuilder};

pub const WINDOW_LABEL: &str = "main";
pub const DASHBOARD_URL: &str = "http://127.0.0.1:8430/";

pub fn window_title_from(user: Option<&str>, host: Option<&str>) -> String {
    match (user, host) {
        (Some(u), Some(h)) if !u.is_empty() && !h.is_empty() => {
            format!("{u}@{h} — Sysmon")
        }
        (_, Some(h)) if !h.is_empty() => format!("{h} — Sysmon"),
        _ => "Sysmon".to_string(),
    }
}

pub fn hostname() -> Option<String> {
    let mut buf = [0u8; 256];
    let rc = unsafe {
        libc::gethostname(buf.as_mut_ptr() as *mut libc::c_char, buf.len())
    };
    if rc != 0 {
        return None;
    }
    let end = buf.iter().position(|&b| b == 0).unwrap_or(buf.len());
    String::from_utf8(buf[..end].to_vec())
        .ok()
        .filter(|s| !s.is_empty())
}

pub fn window_title() -> String {
    let user = std::env::var("USER").ok();
    window_title_from(user.as_deref(), hostname().as_deref())
}

pub fn create_main_window(app: &AppHandle, hidden: bool) -> tauri::Result<()> {
    let url = DASHBOARD_URL.parse().expect("platná URL dashboardu");
    WebviewWindowBuilder::new(app, WINDOW_LABEL, WebviewUrl::External(url))
        .title(window_title())
        .inner_size(1180.0, 820.0)
        .min_inner_size(360.0, 560.0)
        .visible(!hidden)
        .build()?;
    Ok(())
}

pub fn show_main_window(app: &AppHandle) {
    if let Some(window) = app.get_webview_window(WINDOW_LABEL) {
        let _ = window.unminimize();
        let _ = window.show();
        let _ = window.set_focus();
    }
}
```

```rust
use std::collections::VecDeque;
use std::io::{BufRead, BufReader};
use std::os::unix::process::CommandExt;
use std::path::Path;
use std::process::{Child, Command, Stdio};
use std::sync::{Arc, Mutex};
use std::thread;
use std::time::{Duration, Instant};

use tauri::{AppHandle, Manager};

pub const PORT: u16 = 8430;
const READY_TIMEOUT: Duration = Duration::from_secs(10);
const READY_POLL: Duration = Duration::from_millis(50);
const STOP_GRACE: Duration = Duration::from_secs(5);

pub struct SidecarHandle {
    pub owned: bool,
    pub child: Mutex<Option<Child>>,
    pub stderr: Arc<Mutex<VecDeque<String>>>,
}

pub fn port_open() -> bool {
    use std::net::{SocketAddr, TcpStream};
    let addr = SocketAddr::from(([127, 0, 0, 1], PORT));
    TcpStream::connect_timeout(&addr, Duration::from_millis(200)).is_ok()
}

pub fn daemon_dir_ok(dir: &Path) -> bool {
    dir.join("sysmon_app").join("__main__.py").is_file()
}

pub fn ensure_daemon(app: AppHandle) -> Result<SidecarHandle, String> {
    if port_open() {
        return Ok(SidecarHandle {
            owned: false,
            child: Mutex::new(None),
            stderr: Arc::new(Mutex::new(VecDeque::new())),
        });
    }

    let cwd = std::env::current_dir()
        .map_err(|e| format!("Aktuální adresář nejde zjistit: {e}"))?;
    if !daemon_dir_ok(&cwd) {
        return Err(format!(
            "Nenašel jsem démona: v {} chybí sysmon_app/__main__.py.\n\
             Spusť aplikaci z kořene repa sysmon-app.",
            cwd.display()
        ));
    }

    let stderr = Arc::new(Mutex::new(VecDeque::new()));
    let mut child = spawn_daemon(&cwd, &stderr).map_err(|e| {
        format!("Démona se nepodařilo spustit: {e}")
    })?;

    if let Err(err) = wait_ready(&mut child, &stderr) {
        if port_open() {
            // Někdo jiný mezitím obsadil port — použij jeho démona.
            return Ok(SidecarHandle {
                owned: false,
                child: Mutex::new(None),
                stderr,
            });
        }
        return Err(err);
    }

    let _ = app;
    Ok(SidecarHandle {
        owned: true,
        child: Mutex::new(Some(child)),
        stderr,
    })
}

fn spawn_daemon(
    cwd: &Path,
    stderr: &Arc<Mutex<VecDeque<String>>>,
) -> std::io::Result<Child> {
    let mut cmd = Command::new("python3");
    cmd.arg("-m")
        .arg("sysmon_app")
        .current_dir(cwd)
        .stdin(Stdio::null())
        .stdout(Stdio::null())
        .stderr(Stdio::piped());

    unsafe {
        cmd.pre_exec(|| {
            if libc::prctl(libc::PR_SET_PDEATHSIG, libc::SIGTERM) != 0 {
                return Err(std::io::Error::last_os_error());
            }
            if libc::getppid() == 1 {
                libc::_exit(1);
            }
            Ok(())
        });
    }

    let mut child = cmd.spawn()?;
    let pipe = child.stderr.take().expect("stderr je pipovaný");
    let buffer = stderr.clone();
    thread::spawn(move || {
        for line in BufReader::new(pipe).lines().map_while(Result::ok) {
            push_stderr(&buffer, line);
        }
    });
    Ok(child)
}

fn push_stderr(buffer: &Arc<Mutex<VecDeque<String>>>, line: String) {
    let mut buf = buffer.lock().unwrap();
    if buf.len() == 40 {
        buf.pop_front();
    }
    buf.push_back(line);
}

fn stderr_tail(buffer: &Arc<Mutex<VecDeque<String>>>) -> String {
    buffer.lock().unwrap().iter().cloned().collect::<Vec<_>>().join("\n")
}

fn wait_ready(
    child: &mut Child,
    stderr: &Arc<Mutex<VecDeque<String>>>,
) -> Result<(), String> {
    let deadline = Instant::now() + READY_TIMEOUT;
    while Instant::now() < deadline {
        if port_open() {
            return Ok(());
        }
        if let Some(status) = child.try_wait().map_err(|e| e.to_string())? {
            return Err(format!(
                "Démon skončil (kód {}).\n{}",
                status.code().unwrap_or(-1),
                stderr_tail(stderr)
            ));
        }
        thread::sleep(READY_POLL);
    }
    let _ = child.kill();
    let _ = child.wait();
    Err(format!(
        "Démon se do {} s nepřipojil na port {}.\n{}",
        READY_TIMEOUT.as_secs(),
        PORT,
        stderr_tail(stderr)
    ))
}

pub fn stop_owned(app: &AppHandle) {
    let Some(handle) = app.try_state::<SidecarHandle>() else {
        return;
    };
    if !handle.owned {
        return;
    }
    let mut slot = handle.child.lock().unwrap();
    let Some(mut child) = slot.take() else {
        return;
    };
    let pid = child.id() as i32;
    unsafe {
        libc::kill(pid, libc::SIGTERM);
    }
    let deadline = Instant::now() + STOP_GRACE;
    loop {
        match child.try_wait() {
            Ok(Some(_)) => return,
            Ok(None) if Instant::now() < deadline => {
                thread::sleep(Duration::from_millis(50));
            }
            _ => {
                let _ = child.kill();
                let _ = child.wait();
                return;
            }
        }
    }
}
```

```rust
pub const PORT: u16 = 8430;
const READY_TIMEOUT: Duration = Duration::from_secs(10);
const READY_POLL: Duration = Duration::from_millis(50);
const STOP_GRACE: Duration = Duration::from_secs(5);
const UPTIME_RESET: Duration = Duration::from_secs(60);

pub struct SidecarHandle {
    pub owned: bool,
    pub pid: Arc<AtomicI32>,
    pub quitting: Arc<AtomicBool>,
    pub stderr: Arc<Mutex<VecDeque<String>>>,
}

pub fn port_open() -> bool {
    use std::net::{SocketAddr, TcpStream};
    let addr = SocketAddr::from(([127, 0, 0, 1], PORT));
    TcpStream::connect_timeout(&addr, Duration::from_millis(200)).is_ok()
}

pub fn daemon_dir_ok(dir: &Path) -> bool {
    dir.join("sysmon_app").join("__main__.py").is_file()
}

pub fn backoff_for(attempt: usize) -> Option<Duration> {
    match attempt {
        1 => Some(Duration::from_secs(1)),
        2 => Some(Duration::from_secs(2)),
        3 => Some(Duration::from_secs(4)),
        _ => None,
    }
}

pub fn should_restart(exit_code: Option<i32>, quitting: bool) -> bool {
    !quitting && exit_code != Some(2)
}

pub fn ensure_daemon(app: AppHandle) -> Result<SidecarHandle, String> {
    let stderr = Arc::new(Mutex::new(VecDeque::new()));
    let pid = Arc::new(AtomicI32::new(0));
    let quitting = Arc::new(AtomicBool::new(false));

    if port_open() {
        return Ok(SidecarHandle {
            owned: false,
            pid,
            quitting,
            stderr,
        });
    }

    let cwd = std::env::current_dir()
        .map_err(|e| format!("Aktuální adresář nejde zjistit: {e}"))?;
    if !daemon_dir_ok(&cwd) {
        return Err(format!(
            "Nenašel jsem démona: v {} chybí sysmon_app/__main__.py.\n\
             Spusť aplikaci z kořene repa sysmon-app.",
            cwd.display()
        ));
    }

    let mut child = spawn_daemon(&cwd, &stderr)?;

    if let Err(err) = wait_ready(&mut child, &stderr) {
        if port_open() {
            return Ok(SidecarHandle {
                owned: false,
                pid,
                quitting,
                stderr,
            });
        }
        return Err(err);
    }

    pid.store(child.id() as i32, Ordering::SeqCst);
    let handle = SidecarHandle {
        owned: true,
        pid: pid.clone(),
        quitting: quitting.clone(),
        stderr: stderr.clone(),
    };
    start_supervisor(app, cwd, child, pid, quitting, stderr);
    Ok(handle)
}

fn start_supervisor(
    app: AppHandle,
    cwd: PathBuf,
    child: Child,
    pid: Arc<AtomicI32>,
    quitting: Arc<AtomicBool>,
    stderr: Arc<Mutex<VecDeque<String>>>,
) {
    thread::spawn(move || supervise(app, cwd, child, pid, quitting, stderr));
}

fn supervise(
    app: AppHandle,
    cwd: PathBuf,
    mut child: Child,
    pid: Arc<AtomicI32>,
    quitting: Arc<AtomicBool>,
    stderr: Arc<Mutex<VecDeque<String>>>,
) {
    let mut attempt = 0usize;
    let mut started = Instant::now();

    loop {
        let status = child.wait();
        if quitting.load(Ordering::SeqCst) {
            return;
        }
        if started.elapsed() >= UPTIME_RESET {
            attempt = 0;
        }
        let code = status.ok().and_then(|s| s.code());

        if should_restart(code, false) {
            attempt += 1;
            match backoff_for(attempt) {
                Some(delay) => {
                    thread::sleep(delay);
                    if quitting.load(Ordering::SeqCst) {
                        return;
                    }
                }
                None => {
                    if !retry_dialog(&app, &stderr) {
                        return;
                    }
                    attempt = 0;
                }
            }
        } else if !retry_dialog(&app, &stderr) {
            return;
        } else {
            attempt = 0;
        }

        loop {
            if quitting.load(Ordering::SeqCst) {
                return;
            }
            match spawn_daemon(&cwd, &stderr).and_then(|mut next| {
                wait_ready(&mut next, &stderr)?;
                Ok(next)
            }) {
                Ok(next) => {
                    pid.store(next.id() as i32, Ordering::SeqCst);
                    started = Instant::now();
                    child = next;
                    break;
                }
                Err(err) => {
                    push_stderr(&stderr, err);
                    if !retry_dialog(&app, &stderr) {
                        return;
                    }
                    attempt = 0;
                }
            }
        }
    }
}

fn retry_dialog(app: &AppHandle, stderr: &Arc<Mutex<VecDeque<String>>>) -> bool {
    let message = format!(
        "Démon se nepodařilo udržet v chodu.\n\n{}",
        stderr_tail(stderr)
    );
    let (tx, rx) = std::sync::mpsc::channel();
    let handle = app.clone();
    let queued = app.run_on_main_thread(move || {
        let retry = ui::error_dialog(&handle, &message);
        let _ = tx.send(retry);
    });
    if queued.is_err() {
        return false;
    }
    rx.recv_timeout(Duration::from_secs(300)).unwrap_or(false)
}

fn spawn_daemon(
    cwd: &Path,
    stderr: &Arc<Mutex<VecDeque<String>>>,
) -> Result<Child, String> {
    let mut cmd = Command::new("python3");
    cmd.arg("-m")
        .arg("sysmon_app")
        .current_dir(cwd)
        .stdin(Stdio::null())
        .stdout(Stdio::null())
        .stderr(Stdio::piped());

    unsafe {
        cmd.pre_exec(|| {
            if libc::prctl(libc::PR_SET_PDEATHSIG, libc::SIGTERM) != 0 {
                return Err(std::io::Error::last_os_error());
            }
            if libc::getppid() == 1 {
                libc::_exit(1);
            }
            Ok(())
        });
    }

    let mut child = cmd
        .spawn()
        .map_err(|e| format!("Démona se nepodařilo spustit: {e}"))?;
    let pipe = child.stderr.take().expect("stderr je pipovaný");
    let buffer = stderr.clone();
    thread::spawn(move || {
        for line in BufReader::new(pipe).lines().map_while(Result::ok) {
            push_stderr(&buffer, line);
        }
    });
    Ok(child)
}

fn push_stderr(buffer: &Arc<Mutex<VecDeque<String>>>, line: String) {
    let mut buf = buffer.lock().unwrap();
    if buf.len() == 40 {
        buf.pop_front();
    }
    buf.push_back(line);
}

fn stderr_tail(buffer: &Arc<Mutex<VecDeque<String>>>) -> String {
    buffer.lock().unwrap().iter().cloned().collect::<Vec<_>>().join("\n")
}

fn wait_ready(
    child: &mut Child,
    stderr: &Arc<Mutex<VecDeque<String>>>,
) -> Result<(), String> {
    let deadline = Instant::now() + READY_TIMEOUT;
    while Instant::now() < deadline {
        if port_open() {
            return Ok(());
        }
        if let Some(status) = child.try_wait().map_err(|e| e.to_string())? {
            return Err(format!(
                "Démon skončil (kód {}).\n{}",
                status.code().unwrap_or(-1),
                stderr_tail(stderr)
            ));
        }
        thread::sleep(READY_POLL);
    }
    let _ = child.kill();
    let _ = child.wait();
    Err(format!(
        "Démon se do {} s nepřipojil na port {}.\n{}",
        READY_TIMEOUT.as_secs(),
        PORT,
        stderr_tail(stderr)
    ))
}

pub fn stop_owned(app: &AppHandle) {
    let Some(handle) = app.try_state::<SidecarHandle>() else {
        return;
    };
    if !handle.owned {
        return;
    }
    handle.quitting.store(true, Ordering::SeqCst);
    let pid = handle.pid.load(Ordering::SeqCst);
    if pid <= 0 {
        return;
    }
    unsafe {
        libc::kill(pid, libc::SIGTERM);
    }
    let deadline = Instant::now() + STOP_GRACE;
    while Instant::now() < deadline {
        if unsafe { libc::kill(pid, 0) } != 0 {
            return;
        }
        thread::sleep(Duration::from_millis(50));
    }
    unsafe {
        libc::kill(pid, libc::SIGKILL);
    }
}
```

```rust
pub fn create_tray(app: &AppHandle) -> tauri::Result<()> {
    let open = MenuItem::with_id(app, "open", "Otevřít dashboard", true, None::<&str>)?;
    let quit = MenuItem::with_id(app, "quit", "Ukončit", true, None::<&str>)?;
    let menu = Menu::with_items(app, &[&open, &quit])?;

    let mut builder = TrayIconBuilder::with_id("main")
        .menu(&menu)
        .tooltip("Sysmon")
        .on_menu_event(|app, event| match event.id.as_ref() {
            "open" => show_main_window(app),
            "quit" => app.exit(0),
            _ => {}
        })

            if let TrayIconEvent::Click {
                button: MouseButton::Left,
                button_state: MouseButtonState::Up,
                ..
            } = event
            {
                show_main_window(tray.app_handle());
            }
        });

    if let Some(icon) = app.default_window_icon().cloned() {
        builder = builder.icon(icon);
    }
    builder.build(app)?;
    Ok(())
}
```

```rust
pub fn config_dir_from(xdg: Option<&OsStr>, home: Option<&OsStr>) -> PathBuf {
    if let Some(xdg) = xdg {
        let candidate = PathBuf::from(xdg);
        if candidate.is_absolute() {
            return candidate;
        }
    }
    let home = home.map(PathBuf::from).unwrap_or_default();
    home.join(".config")
}

pub fn config_dir() -> PathBuf {
    let xdg = std::env::var_os("XDG_CONFIG_HOME");
    let home = std::env::var_os("HOME");
    config_dir_from(xdg.as_deref(), home.as_deref())
}

pub fn marker_path() -> PathBuf {
    config_dir().join("sysmon-app").join("autostart-asked")
}

pub fn desktop_path() -> PathBuf {
    config_dir().join("autostart").join("sysmon-app.desktop")

pub fn should_offer(desktop_exists: bool, marker_exists: bool) -> bool {
    !desktop_exists && !marker_exists
}

pub fn quote_exec(path: &Path) -> String {
    let raw = path.to_string_lossy();
    let reserved = |c: char| {
        c.is_whitespace() || "\"'\\><~|&;$*?#()`".contains(c)
    };
    if !raw.chars().any(reserved) {
        return raw.into_owned();
    }
    let escaped = raw
        .replace('\\', "\\\\")
        .replace('"', "\\\"")
        .replace('`', "\\`")
        .replace('$', "\\$");
    format!("\"{escaped}\"")
}

         Exec={} --hidden\n\

         X-GNOME-Autostart-enabled=true\n",

}

pub fn maybe_offer(app: &AppHandle) {
    let desktop = desktop_path();
    let marker = marker_path();

    if !should_offer(desktop.exists(), marker.exists()) {
        if desktop.exists() && !marker.exists() {
            write_marker(&marker);
        }
        return;
    }

    if ui::ask_autostart(app) {
        if let Ok(exe) = std::env::current_exe() {
            if let Some(parent) = desktop.parent() {
                let _ = fs::create_dir_all(parent);
            }
            let _ = fs::write(&desktop, desktop_entry(&exe));
        }
    }
    write_marker(&marker);
}

fn write_marker(marker: &Path) {
    if let Some(parent) = marker.parent() {
        let _ = fs::create_dir_all(parent);
    }
    let _ = fs::write(marker, b"");
}
```

```rust
use tauri::Manager;

fn main() {
    let hidden = autostart::hidden_from_args(std::env::args());
```
